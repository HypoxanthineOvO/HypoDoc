\NewDocumentEnvironment{ThemeBlock}{m m +b}{%
  \begingroup
  \def\ThisBlockColor{NatureSand}%
  \ifstrequal{#1}{note}{\def\ThisBlockColor{NatureLilac}}{}%
  \ifstrequal{#1}{objective}{\def\ThisBlockColor{NatureSky}}{}%
  \ifstrequal{#1}{tip}{\def\ThisBlockColor{NatureSand}}{}%
  \ifstrequal{#1}{warning}{\def\ThisBlockColor{NatureEarth}}{}%
  \ifstrequal{#1}{task}{\def\ThisBlockColor{NatureTeal}}{}%
  \ifstrequal{#1}{info}{\def\ThisBlockColor{NatureSage}}{}%
  \begin{tcolorbox}[
    enhanced,arc=1.5mm,boxrule=0pt,
    colback=\ThisBlockColor!18!white,colbacktitle=\ThisBlockColor,
    coltext=ThemeInk,coltitle=ThemeInk,
    fonttitle=\usebeamerfont{block title},fontupper=\usebeamerfont{block body},
    title={#2},left=1.5mm,right=1.5mm,top=0.8mm,bottom=0.8mm,
    toptitle=0.6mm,bottomtitle=0.6mm,before skip=1.2mm,after skip=1.4mm,
    shadow={0.5mm}{-0.5mm}{0mm}{ThemeInk!12!white,opacity=0.4}]
    #3
  \end{tcolorbox}
  \endgroup
}{}

\newcommand{\ExperimentTitle}{%
  \begingroup\usebackgroundtemplate{}%
  \begin{frame}[plain]
    \begin{tikzpicture}[remember picture,overlay]
      \node[inner sep=0pt,opacity=0.55] at (current page.center)
        {\slatecover{\HypoResourceDir/assets/nature-cover.png}};
      \node[align=center,text width=0.76\paperwidth,inner sep=4mm,
        draw=ThemeAccent!80,fill=white,fill opacity=0.86,text opacity=1,
        text=ThemeInk,line width=0.45pt]
        at ([yshift=-0.27\paperheight]current page.north)
        {{\usebeamerfont{title}\inserttitle}\\[6pt]
         {\usebeamerfont{subtitle}\insertsubtitle}};
      \node[text=ThemeInk] at ([yshift=-0.46\paperheight]current page.north) {\insertauthor};
      \node[text=ThemeInk] at ([yshift=-0.54\paperheight]current page.north) {\insertinstitute};
      \node[text=ThemeInk] at ([yshift=-0.87\paperheight]current page.north) {\insertdate};
      \node[text=ThemeInk,font=\bfseries\large] at ([yshift=-0.96\paperheight]current page.north) {HypoDoc};
    \end{tikzpicture}
  \end{frame}\endgroup}
