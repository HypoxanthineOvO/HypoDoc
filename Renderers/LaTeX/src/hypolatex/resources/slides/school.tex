% Bounded, vertically centred navigation. Four or fewer chapters get equal
% cells and linked page dots; larger decks get a current-chapter breadcrumb.
% Read the previous pass's .nav, so this also works for raw exported TeX.
\makeatletter
\newcount\SchoolSections
\newcount\SchoolSlides
\newdimen\SchoolCellWidth
\newcommand{\SchoolCountSections}{%
  \SchoolSections=0
  \begingroup
    \def\sectionentry##1##2##3##4##5{%
      \ifnum##5=\c@part\global\advance\SchoolSections by1\fi}%
    \def\slideentry##1##2##3##4##5##6{}%
    \dohead
  \endgroup}
\newcommand{\SchoolSectionDots}[1]{%
  \begingroup
    \SchoolSlides=0
    \def\sectionentry##1##2##3##4##5{}%
    \def\slideentry##1##2##3##4##5##6{%
      \ifnum##6=\c@part\ifnum##1=#1\advance\SchoolSlides by1\fi\fi}%
    \dohead
    \ifnum\SchoolSlides>12
      {\color{SchoolNavMuted}\the\SchoolSlides\ 页}%
    \else
      \def\slideentry##1##2##3##4##5##6{%
        \ifnum##6=\c@part\ifnum##1=#1
          \beamer@link(##4){%
            \ifnum\c@section=##1\ifnum\c@subsection=##2\ifnum\c@subsectionslide=##3
              \color{ThemeAccent}\else\color{ThemeAccent!45!white}\fi
            \else\color{ThemeAccent!45!white}\fi
            \else\color{ThemeAccent!45!white}\fi
            \ensuremath{\circ}}\kern1pt
        \fi\fi}%
      \dohead
    \fi
  \endgroup}
\newcommand{\SchoolNavigation}{%
  \SchoolCountSections
  \usebeamerfont{section in head/foot}%
  \ifnum\SchoolSections>4
    \hbox to\dimexpr\paperwidth-12mm\relax{%
      \ifnum\c@section>0
        \truncate{0.72\paperwidth}{\textbf{\insertsectionhead}}%
      \else\textbf{报告概览}\fi
      \hfill\the\c@section\,/\,\the\SchoolSections\ 章节}%
  \else
    \ifnum\SchoolSections>0
      \SchoolCellWidth=\dimexpr\paperwidth-12mm\relax
      \divide\SchoolCellWidth by\SchoolSections
      \begingroup
        \def\slideentry##1##2##3##4##5##6{}%
        \def\sectionentry##1##2##3##4##5{%
          \ifnum##5=\c@part
            \parbox[c]{\SchoolCellWidth}{\centering\parskip=0pt
              \ifnum\c@section=##1\color{ThemeAccent}\bfseries
              \else\color{SchoolNavMuted}\mdseries\fi
              \hyperlink{Navigation##3}{\truncate{\dimexpr\SchoolCellWidth-4mm\relax}{\strut##2}}\par
              \vspace{0.3mm}\mdseries\SchoolSectionDots{##1}\par}%
          \fi}%
        \hbox{\dohead}%
      \endgroup
    \else
      \hbox to\dimexpr\paperwidth-12mm\relax{\truncate{0.8\paperwidth}{\insertshorttitle}\hfil}%
    \fi
  \fi}
\setbeamertemplate{headline}{%

\setbeamertemplate{frametitle}{%
  \nointerlineskip
  \begin{beamercolorbox}[wd=\paperwidth,leftskip=9mm,rightskip=9mm]{frametitle}%
    \vskip1.8mm\usebeamerfont{frametitle}\strut\insertframetitle\strut\par
    \ifx\insertframesubtitle\@empty\else
      {\usebeamerfont{framesubtitle}\insertframesubtitle\par}%
    \fi
    \vskip1.8mm
  \end{beamercolorbox}}
% Use explicit point sizes and physical row heights, not ex units inherited
% from a different font. Each field owns its width; long metadata is ellipsised.
\newcommand{\SchoolFooterRow}[2]{%
  \begin{beamercolorbox}[wd=\paperwidth]{#1}%
    \vbox to4mm{\vfil\hbox{\hspace*{6mm}%
      \fontsize{6.5}{8}\selectfont\strut#2\hspace*{6mm}}\vfil}%
  \end{beamercolorbox}}
\setbeamertemplate{footline}{%
  \SchoolFooterRow{author in head/foot}{%
    \hbox to\dimexpr\paperwidth-12mm\relax{%
      \truncate{0.34\paperwidth}{\insertshortauthor}\hfill
      \truncate{0.54\paperwidth}{\insertshortinstitute}}}%
  \SchoolFooterRow{title in head/foot}{%
    \hbox to\dimexpr\paperwidth-12mm\relax{%
      \truncate{0.53\paperwidth}{\insertshorttitle}\hfill
      \truncate{0.23\paperwidth}{\insertshortdate}\hspace{4mm}%
      \insertframenumber\,/\,\inserttotalframenumber}}}
\makeatother
